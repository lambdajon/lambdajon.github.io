data SomeInline where
  MkInline :: Inline k -> SomeInline

data InlineK
  = PlainK
  | BoldK
  | ItalicK

data Inline (k :: InlineK) where
  Plain :: Text -> Inline 'PlainK
  Bold :: [SomeInline] -> Inline 'BoldK
  Italic :: [SomeInline] -> Inline 'ItalicK

pBold :: Parser Bold
pBold = Bold <$> (string "**" *> manyTill pSomeInline (string "**"))

pItalic :: Parser Italic
pItalic = Italic <$> (string "*" *> manyTill pSomeInline (string "*"))

pSomeInline :: Parser SomeInline
pSomeInline = (MkInline <$> pBold) <|> (MkInline <$> pItalic)
